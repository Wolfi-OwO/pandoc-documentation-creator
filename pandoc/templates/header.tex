% ########
% Header Includes
% ---
% Small LaTeX helpers loaded into the document preamble via header-includes.
%
% The main template (custom-template.tex) already loads the heavy packages
% (amsmath, microtype, fancyvrb, tikz, ...) and defines the document styling.
% This file therefore only adds a couple of guarded helper macros. Everything
% here is idempotent: it is safe to load more than once, which matters because
% the build pipeline injects it both through `-H` and through the defaults
% `header-includes` entry.

% ########
% Diagram Imports
% ---
% `import` lets diagram assets pull in generated sub-files by relative path.
\usepackage{import}

% ########
% Custom Helper Commands
% ---
% Defined with \providecommand so they never clash with the main template.
\providecommand{\highlight}[1]{\textcolor{red}{#1}}
\providecommand{\code}[1]{\texttt{\small #1}}
