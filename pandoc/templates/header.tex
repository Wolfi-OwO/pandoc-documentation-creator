########
% Header Includes
% ---
% Shared LaTeX helpers loaded into the document preamble.

########
% Typography
% ---
% Improve spacing and line breaking.
\usepackage{microtype}

########
% Math Packages
% ---
% Additional math support used by generated documents.
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{amsfonts}
\usepackage{mathtools}

########
% Code Styling
% ---
% Basic framed code environment helpers.
\usepackage{framed}

########
% Code Environment
% ---
% Provide a simple framed code environment.
\newenvironment{code}{\begin{framed}\small}{\end{framed}}

########
% PlantUML Support
% ---
% TikZ and import support for diagram assets.
\usepackage{tikz}
\usepackage{import}

########
% Custom Commands
% ---
% Helper commands used by examples and generated output.
\newcommand{\highlight}[1]{\textcolor{red}{#1}}
\newcommand{\code}[1]{\texttt{\small #1}}

########
% Verbatim Blocks
% ---
% Improve the default verbatim environment.
\usepackage{fancyvrb}
\RecustomVerbatimEnvironment{Verbatim}{Verbatim}{
  frame=lines,
  framesep=5pt,
  samepage=false,
  commandchars=\\\{\}
}

########
% Title Page Styling
% ---
% Fallback `\maketitle` layout.
\renewcommand{\maketitle}{%
  \begin{titlepage}%
    \centering%
    {\Large\bfseries\@title\par}%
    \vspace{1.5cm}%
    {\large\@author\par}%
    \vspace{1.5cm}%
    {\large\@date\par}%
    \vfill%
  \end{titlepage}%
}
